Step 2 — Calculator Class (Core Logic)

JavaCalculator.java
public class Calculator {

    // ── Basic operations ─────────────────────────────────────
    public double add(double a, double b) {
        return a + b;
    }

    public double subtract(double a, double b) {
        return a - b;
    }

    public double multiply(double a, double b) {
        return a * b;
    }

    public double divide(double a, double b) {
        if (b == 0) {
            throw new DivisionByZeroException(a);
        }
        return a / b;
    }

    // ── Route by operation string ─────────────────────────────
    public double calculate(double a, String op, double b) {
        switch (op) {
            case "+": return add(a, b);
            case "-": return subtract(a, b);
            case "*": return multiply(a, b);
            case "/": return divide(a, b);
            default:
                throw new IllegalArgumentException("Unknown operator: " + op);
        }
    }

    // ── Describe the operation ────────────────────────────────
    public String getOperationName(String op) {
        switch (op) {
            case "+": return "Addition";
            case "-": return "Subtraction";
            case "*": return "Multiplication";
            case "/": return "Division";
            default:  return "Unknown";
        }
    }
}

Step 3 — CalculationRecord Class

JavaCalculationRecord.java
// Stores a single completed calculation — Day 4 OOP
public class CalculationRecord {

    private final String operation;
    private final double a;
    private final double b;
    private final double result;
    private final boolean success;

    // Constructor for successful calculation
    public CalculationRecord(double a, String operation, double b, double result) {
        this.a         = a;
        this.operation = operation;
        this.b         = b;
        this.result    = result;
        this.success   = true;
    }

    // Constructor for failed calculation
    public CalculationRecord(double a, String operation, double b) {
        this.a         = a;
        this.operation = operation;
        this.b         = b;
        this.result    = 0;
        this.success   = false;
    }

    // Getters
    public String  getOperation() { return operation; }
    public double  getResult()    { return result; }
    public boolean isSuccess()    { return success; }

    @Override
    public String toString() {
        if (!success) {
            return String.format("  [ERROR] %.1f %s %.1f = FAILED", a, operation, b);
        }
        return String.format("  [OK]    %.1f %s %.1f = %.2f", a, operation, b, result);
    }
}

Step 4 — CalculatorHistory Class

JavaCalculatorHistory.java
import java.util.*;

public class CalculatorHistory {

    // ArrayList of objects — Day 6
    private final List<CalculationRecord> records = new ArrayList<>();

    public void add(CalculationRecord r) {
        records.add(r);
    }

    public int size() { return records.size(); }

    // Print all records
    public void printAll() {
        if (records.isEmpty()) {
            System.out.println("  No history yet.");
            return;
        }
        for (CalculationRecord r : records) {
            System.out.println(r);  // calls toString()
        }
    }

    // Stats — only from successful calculations
    public void printStats() {
        List<Double> results = new ArrayList<>();
        for (CalculationRecord r : records) {
            if (r.isSuccess()) results.add(r.getResult());
        }
        if (results.isEmpty()) {
            System.out.println("  No successful calculations.");
            return;
        }
        double max = Collections.max(results);
        double min = Collections.min(results);
        double sum = 0;
        for (double v : results) sum += v;
        double avg = sum / results.size();

        System.out.printf("  Results: %d successful, %d total%n",
            results.size(), records.size());
        System.out.printf("  Highest : %.2f%n", max);
        System.out.printf("  Lowest  : %.2f%n", min);
        System.out.printf("  Average : %.2f%n", avg);
    }

    // Operation frequency — HashMap Day 6
    public void printOpCounts() {
        Map<String, Integer> counts = new HashMap<>();
        for (CalculationRecord r : records) {
            String op = r.getOperation();
            counts.put(op, counts.getOrDefault(op, 0) + 1);
        }
        for (Map.Entry<String, Integer> e : counts.entrySet()) {
            System.out.printf("  %-4s used %d time(s)%n", e.getKey(), e.getValue());
        }
    }
}

Step 5 — AdvancedCalculator (Inheritance)

JavaAdvancedCalculator.java
// Inherits all Calculator methods + adds new ones — Day 5
public class AdvancedCalculator extends Calculator {

    public double power(double base, double exp) {
        return Math.pow(base, exp);
    }

    public double sqrt(double n) {
        if (n < 0) {
            throw new IllegalArgumentException(
                "Cannot take sqrt of negative number: " + n);
        }
        return Math.sqrt(n);
    }

    public double modulo(double a, double b) {
        if (b == 0) {
            throw new DivisionByZeroException(a);
        }
        return a % b;
    }

    // Override calculate to also handle advanced ops
    @Override
    public double calculate(double a, String op, double b) {
        switch (op) {
            case "^": return power(a, b);
            case "%": return modulo(a, b);
            default:  return super.calculate(a, op, b);  // delegate to parent
        }
    }
}

Step 6 — Main Runner

JavaMain.java
public class Main {

    // Helper: run a calculation + record it + print it
    static void run(Calculator calc, CalculatorHistory hist,
                   double a, String op, double b) {
        try {
            double result = calc.calculate(a, op, b);
            hist.add(new CalculationRecord(a, op, b, result));
            System.out.printf("  %.1f %s %.1f = %.2f%n", a, op, b, result);
        } catch (DivisionByZeroException e) {
            hist.add(new CalculationRecord(a, op, b));
            System.out.println("  [ERROR] " + e.getMessage());
        } catch (IllegalArgumentException e) {
            hist.add(new CalculationRecord(a, op, b));
            System.out.println("  [ERROR] " + e.getMessage());
        }
    }

    public static void main(String[] args) {

        AdvancedCalculator calc = new AdvancedCalculator();
        CalculatorHistory  hist = new CalculatorHistory();

        // ── Basic Operations ─────────────────────────────────────
        System.out.println("====== BASIC OPERATIONS ======");
        run(calc, hist, 10, "+", 5);
        run(calc, hist, 20, "-", 8);
        run(calc, hist, 6,  "*", 7);
        run(calc, hist, 15, "/", 3);

        // ── Error Handling ───────────────────────────────────────
        System.out.println("\n====== ERROR HANDLING ======");
        run(calc, hist, 9,  "/", 0);    // DivisionByZeroException
        run(calc, hist, 5,  "?", 3);    // IllegalArgumentException

        // ── Advanced Operations ──────────────────────────────────
        System.out.println("\n====== ADVANCED OPERATIONS ======");
        run(calc, hist, 2,  "^", 8);   // 2^8 = 256
        run(calc, hist, 17, "%", 5);   // 17 mod 5 = 2
        try {
            double sqrtResult = calc.sqrt(144);
            System.out.printf("  sqrt(144) = %.1f%n", sqrtResult);
            hist.add(new CalculationRecord(144, "sqrt", 0, sqrtResult));
        } catch (IllegalArgumentException e) {
            System.out.println("  [ERROR] " + e.getMessage());
        }
        try {
            calc.sqrt(-4);   // negative — throws
        } catch (IllegalArgumentException e) {
            System.out.println("  [ERROR] " + e.getMessage());
        }

        // ── History ──────────────────────────────────────────────
        System.out.println("\n====== CALCULATION HISTORY ======");
        hist.printAll();

        // ── Stats ────────────────────────────────────────────────
        System.out.println("\n====== STATISTICS ======");
        hist.printStats();

        // ── Operation counts ─────────────────────────────────────
        System.out.println("\n====== OPERATION FREQUENCY ======");
        hist.printOpCounts();
    }
}

JavaDay8MiniProject.java — All-in-one version
import java.util.*;

public class Day8MiniProject {

    // ── Custom Exception ─────────────────────────────────────
    static class DivisionByZeroException extends RuntimeException {
        private final double dividend;
        public DivisionByZeroException(double d) {
            super("Cannot divide " + d + " by zero");
            this.dividend = d;
        }
        public double getDividend() { return dividend; }
    }

    // ── CalculationRecord ────────────────────────────────────
    static class CalculationRecord {
        final String  operation;
        final double  a, b, result;
        final boolean success;

        CalculationRecord(double a, String op, double b, double res) {
            this.a = a; this.operation = op; this.b = b;
            this.result = res; this.success = true;
        }
        CalculationRecord(double a, String op, double b) {
            this.a = a; this.operation = op; this.b = b;
            this.result = 0; this.success = false;
        }
        public String toString() {
            if (!success) return String.format("  [ERROR] %.1f %s %.1f = FAILED",a,operation,b);
            return String.format("  [OK]    %.1f %s %.1f = %.2f",a,operation,b,result);
        }
    }

    // ── Calculator ───────────────────────────────────────────
    static class Calculator {
        public double add(double a, double b) { return a + b; }
        public double sub(double a, double b) { return a - b; }
        public double mul(double a, double b) { return a * b; }
        public double div(double a, double b) {
            if (b == 0) throw new DivisionByZeroException(a);
            return a / b;
        }
        public double calculate(double a, String op, double b) {
            switch (op) {
                case "+": return add(a, b);
                case "-": return sub(a, b);
                case "*": return mul(a, b);
                case "/": return div(a, b);
                default:  throw new IllegalArgumentException("Unknown op: " + op);
            }
        }
    }

    // ── AdvancedCalculator ───────────────────────────────────
    static class AdvancedCalculator extends Calculator {
        public double power(double base, double exp) { return Math.pow(base, exp); }
        public double sqrt(double n) {
            if (n < 0) throw new IllegalArgumentException("Negative sqrt: " + n);
            return Math.sqrt(n);
        }
        @Override
        public double calculate(double a, String op, double b) {
            if (op.equals("^")) return power(a, b);
            if (op.equals("%")) { if(b==0) throw new DivisionByZeroException(a); return a%b; }
            return super.calculate(a, op, b);
        }
    }

    // ── CalculatorHistory ────────────────────────────────────
    static class CalculatorHistory {
        private final List<CalculationRecord> records = new ArrayList<>();
        public void add(CalculationRecord r) { records.add(r); }
        public void printAll() { records.forEach(System.out::println); }
        public void printStats() {
            List<Double> res = new ArrayList<>();
            for (CalculationRecord r : records) if (r.success) res.add(r.result);
            if (res.isEmpty()) { System.out.println("  No data"); return; }
            double sum = 0; for (double v : res) sum += v;
            System.out.printf("  Successful: %d / %d%n", res.size(), records.size());
            System.out.printf("  Highest   : %.2f%n", Collections.max(res));
            System.out.printf("  Lowest    : %.2f%n", Collections.min(res));
            System.out.printf("  Average   : %.2f%n", sum / res.size());
        }
        public void printOpCounts() {
            Map<String, Integer> m = new HashMap<>();
            for (CalculationRecord r : records)
                m.put(r.operation, m.getOrDefault(r.operation, 0) + 1);
            List<String> keys = new ArrayList<>(m.keySet());
            Collections.sort(keys);
            keys.forEach(k -> System.out.printf("  %-5s: %d time(s)%n", k, m.get(k)));
        }
    }

    // ── Helper ───────────────────────────────────────────────
    static void run(Calculator c, CalculatorHistory h,
                   double a, String op, double b) {
        try {
            double r = c.calculate(a, op, b);
            h.add(new CalculationRecord(a, op, b, r));
            System.out.printf("  %.1f %s %.1f = %.2f%n", a, op, b, r);
        } catch (DivisionByZeroException e) {
            h.add(new CalculationRecord(a, op, b));
            System.out.println("  [ERROR] " + e.getMessage());
        } catch (IllegalArgumentException e) {
            h.add(new CalculationRecord(a, op, b));
            System.out.println("  [ERROR] " + e.getMessage());
        }
    }

    // ── MAIN ─────────────────────────────────────────────────
    public static void main(String[] args) {
        AdvancedCalculator calc = new AdvancedCalculator();
        CalculatorHistory  hist = new CalculatorHistory();

        System.out.println("====== BASIC OPERATIONS ======");
        run(calc,hist,10,"+",5);
        run(calc,hist,20,"-",8);
        run(calc,hist,6, "*",7);
        run(calc,hist,15,"/",3);

        System.out.println("\n====== ERROR HANDLING ======");
        run(calc,hist,9, "/",0);
        run(calc,hist,5, "?",3);

        System.out.println("\n====== ADVANCED OPERATIONS ======");
        run(calc,hist,2, "^",8);
        run(calc,hist,17,"%",5);
        try {
            double s = calc.sqrt(144);
            System.out.printf("  sqrt(144) = %.1f%n", s);
            hist.add(new CalculationRecord(144,"sqrt",0,s));
        } catch (Exception e) { System.out.println("  [ERROR] "+e.getMessage()); }
        try { calc.sqrt(-4); }
        catch (IllegalArgumentException e) { System.out.println("  [ERROR] "+e.getMessage()); }

        System.out.println("\n====== CALCULATION HISTORY ======");
        hist.printAll();

        System.out.println("\n====== STATISTICS ======");
        hist.printStats();

        System.out.println("\n====== OPERATION FREQUENCY ======");
        hist.printOpCounts();
    }
}

import io.github.bonigarcia.wdm.WebDriverManager;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.chrome.ChromeDriver;

public class FirstSeleniumScript {
    public static void main(String[] args) {

        // Set up ChromeDriver automatically
        WebDriverManager.chromedriver().setup();

        // Create WebDriver object (Chrome browser)
        WebDriver driver = new ChromeDriver();

        try {
            // Navigate to URL
            driver.get("https://opensource-demo.orangehrmlive.com");

            // Get and print page title
            String title = driver.getTitle();
            System.out.println("Page title: " + title);

        } finally {
            driver.quit();  // ALWAYS in finally!
        }
    }
}
