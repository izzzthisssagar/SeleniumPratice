Text Input Interactions

JavaTextInput.java
import org.openqa.selenium.Keys;

WebElement field = driver.findElement(By.name("username"));

// Basic typing
field.clear();
field.sendKeys("Admin");

// Read back what was typed
String value = field.getAttribute("value"); // "Admin"

// Type special keys using Keys enum
field.sendKeys(Keys.TAB);         // press Tab (move focus to next field)
field.sendKeys(Keys.ENTER);       // press Enter (submit form)
field.sendKeys(Keys.BACK_SPACE);  // delete last character
field.sendKeys(Keys.CONTROL, "a"); // Ctrl+A (select all)
field.sendKeys(Keys.CONTROL, "c"); // Ctrl+C (copy)
field.sendKeys(Keys.CONTROL, "v"); // Ctrl+V (paste)
field.sendKeys(Keys.DELETE);       // delete selected text
field.sendKeys(Keys.HOME);         // go to beginning of line
field.sendKeys(Keys.END);          // go to end of line
field.sendKeys(Keys.ARROW_DOWN);   // arrow keys
field.sendKeys(Keys.F5);           // function keys

// Type + press Enter in one call (login shortcut)
field.sendKeys("Admin" + Keys.ENTER);

// Type in a textarea (multi-line)
WebElement textarea = driver.findElement(By.tagName("textarea"));
textarea.clear();
textarea.sendKeys("Line 1\nLine 2\nLine 3");  // \n for new line

// Clear using Ctrl+A then Delete (when clear() doesn't work)
field.sendKeys(Keys.CONTROL, "a");
field.sendKeys(Keys.DELETE);

WebElement checkbox = driver.findElement(By.id("rememberMe"));

// Check current state
boolean isChecked = checkbox.isSelected();

if (!checkbox.isSelected()) {
    checkbox.click();
    System.out.println("Checkbox checked ✓");
}

// Uncheck the box (if currently checked)
if (checkbox.isSelected()) {
    checkbox.click();
    System.out.println("Checkbox unchecked ✓");
}

// Toggle helper method — idempotent
public static void setCheckbox(WebElement checkbox, boolean check) {
    if (checkbox.isSelected() != check) {
        checkbox.click();
    }
}

// All checkboxes on page
List<WebElement> allBoxes = driver.findElements(
    By.cssSelector("input[type='checkbox']"));
System.out.println("Total checkboxes: " + allBoxes.size());

// Select all checkboxes
for (WebElement box : allBoxes) {
    if (!box.isSelected()) box.click();
}

// Count checked boxes
long checkedCount = allBoxes.stream()
    .filter(WebElement::isSelected).count();
System.out.println("Checked: " + checkedCount);

//       <input type="radio" name="gender" value="female"> Female
//       <input type="radio" name="gender" value="other"> Other

// Get all radio options in group
List<WebElement> genderRadios = driver.findElements(By.name("gender"));
System.out.println("Options: " + genderRadios.size()); // 3

// Click specific option by value attribute
for (WebElement radio : genderRadios) {
    if (radio.getAttribute("value").equals("female")) {
        radio.click();
        break;
    }
}

// Get the currently selected radio value
String selectedValue = "";
for (WebElement radio : genderRadios) {
    if (radio.isSelected()) {
        selectedValue = radio.getAttribute("value");
        break;
    }
}
System.out.println("Selected: " + selectedValue); // female

// Select by XPath (most readable)
driver.findElement(By.xpath("//input[@type='radio'][@value='male']")).click();

// Helper method
public static void selectRadio(List<WebElement> options, String value) {
    for (WebElement r : options) {
        if (r.getAttribute("value").equals(value)) {
            r.click();
            return;
        }
    }
    throw new IllegalArgumentException("Radio value not found: " + value);
}

Select Dropdowns — Select Class

Selenium has a dedicated Select class for native HTML <select> elements. Import it and use it — much cleaner than clicking options manually.

JavaSelectDropdown.java
import org.openqa.selenium.support.ui.Select;

// HTML: <select id="country">
//   <option value="np">Nepal</option>
//   <option value="in">India</option>
//   <option value="us">United States</option>
// </select>

WebElement rawDropdown = driver.findElement(By.id("country"));
Select dropdown    = new Select(rawDropdown);

// ── Select options ────────────────────────────────────────────
dropdown.selectByVisibleText("Nepal");    // by the text you see in dropdown
dropdown.selectByValue("in");            // by value attribute
dropdown.selectByIndex(0);              // by position (0 = first)

// ── Get selected option ───────────────────────────────────────
WebElement selected = dropdown.getFirstSelectedOption();
System.out.println("Selected text : " + selected.getText());
System.out.println("Selected value: " + selected.getAttribute("value"));

// ── Get all options ───────────────────────────────────────────
List<WebElement> options = dropdown.getOptions();
System.out.println("Total options: " + options.size());
for (WebElement opt : options) {
    System.out.println("  " + opt.getText() + " (value=" + opt.getAttribute("value") + ")");
}

// ── Check if multiple select ──────────────────────────────────
boolean isMulti = dropdown.isMultiple();
System.out.println("Multi-select: " + isMulti);

// ── Verify option exists before selecting ─────────────────────
boolean hasNepal = options.stream()
    .anyMatch(o -> o.getText().equals("Nepal"));
System.out.println("Has Nepal: " + hasNepal);

Custom Dropdowns (Non-select)

Many modern apps use custom div-based dropdowns instead of native <select>. OrangeHRM uses these. They need a different approach — click to open, then click option.

JavaCustomDropdown.java
// OrangeHRM custom dropdown pattern (div-based, not select)
// HTML: <div class="oxd-select-wrapper">
//   <div class="oxd-select-text">-- Select --</div>
//   <div class="oxd-select-dropdown">
//     <div class="oxd-select-option">Admin</div>
//     <div class="oxd-select-option">User</div>
//   </div>
// </div>

// Step 1: Click to open dropdown
WebElement dropdownTrigger = driver.findElement(
    By.cssSelector(".oxd-select-wrapper"));
dropdownTrigger.click();
Thread.sleep(500); // wait for options to appear

// Step 2: Click the option you want
driver.findElement(By.xpath("//div[@class='oxd-select-option'][text()='Admin']")).click();

// Alternative: click any option by index
List<WebElement> options = driver.findElements(
    By.cssSelector(".oxd-select-option"));
options.get(1).click(); // click 2nd option

// Helper method for custom dropdowns
public static void selectCustomDropdown(WebDriver driver,
        By triggerLocator, By optionLocator, String optionText)
        throws InterruptedException {
    driver.findElement(triggerLocator).click();
    Thread.sleep(500);
    List<WebElement> opts = driver.findElements(optionLocator);
    for (WebElement opt : opts) {
        if (opt.getText().trim().equals(optionText)) {
            opt.click();
            return;
        }
    }
    throw new RuntimeException("Option not found: " + optionText);
}

Element State Checks

JavaElementState.java
// isDisplayed() — visible on screen
boolean visible = element.isDisplayed();
// Returns false if: display:none, visibility:hidden, opacity:0, off-screen
// Throws NoSuchElementException if element not in DOM at all

// isEnabled() — interactable
boolean enabled = element.isEnabled();
// Returns false if HTML has disabled attribute

// isSelected() — for checkboxes and radio buttons
boolean selected = element.isSelected();
// Returns true if checked/selected

// Safe check — is element present in DOM?
boolean isPresent(WebDriver driver, By locator) {
    return !driver.findElements(locator).isEmpty();
}

// Safe check — is element visible?
boolean isVisible(WebDriver driver, By locator) {
    try {
        return driver.findElement(locator).isDisplayed();
    } catch (NoSuchElementException e) {
        return false;
    }
}

// Complete state check before interaction
WebElement btn = driver.findElement(By.id("submitBtn"));
if (btn.isDisplayed() && btn.isEnabled()) {
    btn.click();
    System.out.println("Clicked button");
} else {
    System.out.println("Button not clickable: displayed=" + btn.isDisplayed()
        + " enabled=" + btn.isEnabled());
}

ElementHelper Utility Class

JavaElementHelper.java — utils/ElementHelper.java
package com.qaautomation.utils;

import org.openqa.selenium.*;
import org.openqa.selenium.support.ui.Select;
import java.util.List;

public class ElementHelper {

    // Type into field (clear first)
    public static void type(WebDriver d, By locator, String text) {
        WebElement el = d.findElement(locator);
        el.clear();
        el.sendKeys(text);
    }

    // Click element
    public static void click(WebDriver d, By locator) {
        d.findElement(locator).click();
    }

    // Get text
    public static String getText(WebDriver d, By locator) {
        return d.findElement(locator).getText().trim();
    }

    // Get attribute
    public static String getAttribute(WebDriver d, By locator, String attr) {
        return d.findElement(locator).getAttribute(attr);
    }

    // Is element present?
    public static boolean isPresent(WebDriver d, By locator) {
        return !d.findElements(locator).isEmpty();
    }

    // Is element visible?
    public static boolean isVisible(WebDriver d, By locator) {
        try { return d.findElement(locator).isDisplayed(); }
        catch (NoSuchElementException e) { return false; }
    }

    // Select dropdown by visible text
    public static void selectByText(WebDriver d, By locator, String text) {
        new Select(d.findElement(locator)).selectByVisibleText(text);
    }

    // Select dropdown by value
    public static void selectByValue(WebDriver d, By locator, String value) {
        new Select(d.findElement(locator)).selectByValue(value);
    }

    // Set checkbox state
    public static void setCheckbox(WebDriver d, By locator, boolean check) {
        WebElement cb = d.findElement(locator);
        if (cb.isSelected() != check) cb.click();
    }

    // Get all texts from element list
    public static List<String> getAllTexts(WebDriver d, By locator) {
        List<String> texts = new java.util.ArrayList<>();
        for (WebElement el : d.findElements(locator))
            texts.add(el.getText().trim());
        return texts;
    }
}

Common Mistakes

❌ getText() vs getAttribute("value")

getText() gets visible text between tags. getAttribute("value") gets the value property of form inputs. For inputs, always use getAttribute("value").

❌ Not using Select class

Trying to click options inside a native <select> without the Select wrapper often fails. Always wrap: new Select(driver.findElement(...))

❌ clear() on disabled field

Calling clear() on a disabled or readonly input throws ElementNotInteractableException. Always check isEnabled() first.

❌ Not clearing before sendKeys()

If a field has pre-filled text, sendKeys appends. Always clear() first unless intentionally appending.

❌ Custom dropdown treated as Select

Custom div-based dropdowns cannot use the Select class. They need click() to open + findElement to pick option.

❌ File upload with relative path

fileInput.sendKeys("test.png") fails — Selenium needs an ABSOLUTE path. Use new File("path").getAbsolutePath().
